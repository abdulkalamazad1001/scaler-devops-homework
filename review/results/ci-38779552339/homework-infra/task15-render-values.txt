# Source: notes-chart/templates/deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: review-deploy
  labels:
    helm.sh/chart: notes-chart-0.1.0
    app.kubernetes.io/name: notes-chart
    app.kubernetes.io/instance: review
    app.kubernetes.io/version: "1.0"
    app.kubernetes.io/managed-by: Helm
    environment: development
spec:
  replicas: 1
  selector:
    matchLabels:
      app.kubernetes.io/name: notes-chart
      app.kubernetes.io/instance: review
  template:
    metadata:
      labels:
        app.kubernetes.io/name: notes-chart
        app.kubernetes.io/instance: review
        environment: development
      annotations:
        # Changes whenever the ConfigMap content changes, so pods restart and
        # pick up the new config on helm upgrade.
        checksum/config: fd39a28f669b68c654870afc94d4258bac27a96baee8030bfb26161f7e219130
    spec:
      containers:
        - name: notes
          image: "nginx:1.24"
          imagePullPolicy: IfNotPresent
          ports:
            - name: http
              containerPort: 80
              protocol: TCP
          envFrom:
            - configMapRef:
                name: review-config
          volumeMounts:
            - name: html
              mountPath: /usr/share/nginx/html/index.html
              subPath: index.html
              readOnly: true
          readinessProbe:
            httpGet:
              path: /
              port: http
            initialDelaySeconds: 2
            periodSeconds: 5
          livenessProbe:
            httpGet:
              path: /
              port: http
            initialDelaySeconds: 10
            periodSeconds: 10
          resources:
            limits:
              cpu: 200m
              memory: 128Mi
            requests:
              cpu: 50m
              memory: 32Mi
      volumes:
        - name: html
          configMap:
            name: review-config
            items:
              - key: index.html
                path: index.html
---
